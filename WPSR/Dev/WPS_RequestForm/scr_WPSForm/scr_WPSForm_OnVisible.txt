// Load all related data for this WPS ID
Concurrent(
    // Request Info
    Set(
        varRequestInfo,
        LookUp(
            WPSRequest,
            WPSID = varWPSID
        )
    ),
    // Attachments
    ClearCollect(
        colAttachments,
        Filter(
            WPSAttachments,
            WPSID = varWPSID
        )
    ),
    // Comments
    ClearCollect(
        colComments,
        Filter(
            WPSComments,
            WPSID = varWPSID
        )
    )
);
If(
    varAppMode = "edit" || varAppMode = "approval",
    ClearCollect(
        colWeldDetailsEdit1,
        Filter(
            WPSInternalWelds,
            WPSID = varWPSID
        )
    );
    ClearCollect(
        colWPSOSVWeldsEdit,
        Filter(
            WPSOSVwelds,
            WPSID = varWPSID
        )
    )
)
